% dates | ribbon | content — minipage layout uses almost full line width
\newcommand{\timelineentry}[2]{%
  \noindent
  \begin{minipage}[t]{\datecolwidth}
    \raggedleft\small #1
  \end{minipage}%
  \hspace{0.25cm}%
  \raisebox{-0.05ex}{\ribbon}%
  \hspace{0.3cm}%
  \begin{minipage}[t]{\dimexpr\textwidth-\datecolwidth-0.75cm\relax}
    #2
  \end{minipage}\par
  \vspace{6pt}
}

  \textit{#2}\hfill\textit{#3}\\[2pt]
}
